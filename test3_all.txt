program full_feature_test;

const
    MAX = 5;
    PI = 3.14;
    CH = 'Z';
    STR = "hello from const string";
    OK = true;

struct Point {
    x, y: integer;
    value: real;
    tag: char;
};

function calc(base: integer; scale: real): integer;
var
    temp: integer;
begin
    temp := base + 2;

    if scale > PI then
        temp := temp + 1
    else
        temp := temp - 1;

    return temp
end;

procedure updatePoint(var p: struct Point; var n: integer; valueCopy: integer; ch: char; textCopy: string);
begin
    p.x := valueCopy;
    p.y := n + 1;
    p.value := 3.5;
    p.tag := ch;

    write(textCopy);
    write(p.x);

    n := n + 1
end;

var
    a, b, c: integer;
    x, y: real;
    flag: boolean;
    letter: char;
    msg: string;
    arr: array[5] of integer;
    pt: struct Point;

begin
    a := 0;
    b := 3;
    c := calc(b, PI);

    x := 2.5;
    y := x + c / 2;

    letter := CH;
    msg := "main string literal";

    arr[0] := a;
    arr[1] := b;
    arr[2] := arr[0] + arr[1] * 2;

    pt.x := c;
    pt.y := arr[2];
    pt.value := y;
    pt.tag := letter;

    updatePoint(pt, c, arr[2], letter, STR);

    flag := (c > b) && !(a > b) || false;

    if flag then
        write("flag is true")
    else
        write("flag is false");

    while a < MAX do
    begin
        a := a + 1;
        write(a)
    end;

    write(msg);
    write(letter);
    write(arr[2]);
    write(pt.x);
    write(pt.value);
    write(c)
end.
